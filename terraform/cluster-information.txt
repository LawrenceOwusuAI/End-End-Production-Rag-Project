2. Allow Metrics Server to communicate with kubelets that use self-signed certificates
kubectl patch deployment metrics-server -n kube-system --type=json -p "[{\"op\":\"add\",\"path\":\"/spec/template/spec/containers/0/args/-\",\"value\":\"--kubelet-insecure-tls\"}]"

3. Run Metrics Server on the node network. This fixes API-server-to-Metrics-Server connectivity in our kubeadm + Calico setup
kubectl patch deployment metrics-server -n kube-system --type=strategic -p "{\"spec\":{\"template\":{\"spec\":{\"hostNetwork\":true,\"dnsPolicy\":\"ClusterFirstWithHostNet\"}}}}"


4. Change Metrics Server listening port from 10250 to 4443 because kubelet already uses port 10250 on the node
kubectl patch deployment metrics-server -n kube-system --type=json -p "[{\"op\":\"replace\",\"path\":\"/spec/template/spec/containers/0/args/1\",\"value\":\"--secure-port=4443\"},{\"op\":\"replace\",\"path\":\"/spec/template/spec/containers/0/ports/0/containerPort\",\"value\":4443}]"

       #CREATION OF NAMESPACED USER

# 1. CREATE / VERIFY THE RAG NAMESPACE

# Create the application namespace.
kubectl create namespace rag

# Verify the namespace.
kubectl get namespace rag


# 2. GENERATE OWUSU'S PRIVATE KEY

# Generate a 2048-bit RSA private key for Owusu.
openssl genrsa -out owusu.key 2048


# 3. GENERATE OWUSU'S CERTIFICATE SIGNING REQUEST

# CN=owusu becomes the Kubernetes username.
# O=developers becomes the Kubernetes group.
openssl req -new \
  -key owusu.key \
  -out owusu.csr \
  -subj "/CN=owusu/O=developers"



# 4. ENCODE THE CSR

# Convert the CSR to base64 so Kubernetes can accept it.
CSR=$(base64 -w 0 owusu.csr)

# 5. CREATE THE KUBERNETES CERTIFICATE SIGNING REQUEST

# Create the CSR manifest automatically.
cat > owusu-csr.yaml <<EOF
apiVersion: certificates.k8s.io/v1
kind: CertificateSigningRequest
metadata:
  name: owusu
spec:
  request: ${CSR}
  signerName: kubernetes.io/kube-apiserver-client
  expirationSeconds: 31536000
  usages:
    - client auth
EOF

# Submit the CSR to Kubernetes.
kubectl apply -f owusu-csr.yaml



# 6. CHECK THE CSR

# The request should initially show Pending.
kubectl get csr owusu

# 7. APPROVE OWUSU'S CERTIFICATE


# Approve the user certificate request as cluster administrator.
kubectl certificate approve owusu

# Verify that the certificate was approved and issued.
kubectl get csr owusu


# 8. DOWNLOAD THE SIGNED CERTIFICATE

# Extract the Kubernetes-signed client certificate.
kubectl get csr owusu \
  -o jsonpath='{.status.certificate}' \
  | base64 -d > owusu.crt

# Verify the generated files.
ls -l owusu.key owusu.csr owusu.crt



# 9. CREATE A DEVELOPER ROLE FOR THE RAG NAMESPACE

# Create a namespace-scoped Role.
# The developer can manage application workloads but cannot
# access Kubernetes Secrets or administer cluster nodes.
cat > developer-role.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: Role

metadata:
  name: developer
  namespace: rag

rules:

  # Read pods and pod logs.
  - apiGroups:
      - ""
    resources:
      - pods
      - pods/log
    verbs:
      - get
      - list
      - watch

  # Allow exec and port forwarding to pods.
  - apiGroups:
      - ""
    resources:
      - pods/exec
      - pods/portforward
    verbs:
      - create

  # Manage Services and ConfigMaps.
  - apiGroups:
      - ""
    resources:
      - services
      - configmaps
    verbs:
      - get
      - list
      - watch
      - create
      - update
      - patch
      - delete

  # Manage common application workloads.
  - apiGroups:
      - apps
    resources:
      - deployments
      - replicasets
      - statefulsets
      - daemonsets
    verbs:
      - get
      - list
      - watch
      - create
      - update
      - patch
      - delete

  # Manage Jobs and CronJobs.
  - apiGroups:
      - batch
    resources:
      - jobs
      - cronjobs
    verbs:
      - get
      - list
      - watch
      - create
      - update
      - patch
      - delete
EOF

# Apply the developer Role.
kubectl apply -f developer-role.yaml

# Verify the Role.
kubectl get role developer -n rag


# 10. BIND THE DEVELOPERS GROUP TO THE DEVELOPER ROLE

# Owusu's certificate contains O=developers.
# This RoleBinding gives that group the developer Role.
cat > developer-rolebinding.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding

metadata:
  name: developers
  namespace: rag

subjects:
  - kind: Group
    name: developers
    apiGroup: rbac.authorization.k8s.io

roleRef:
  kind: Role
  name: developer
  apiGroup: rbac.authorization.k8s.io
EOF

# Apply the RoleBinding.
kubectl apply -f developer-rolebinding.yaml

# Verify the RoleBinding.
kubectl get rolebinding developers -n rag

CONTROL_PLANE_EIP="YOUR_CONTROL_PLANE_EIP"



# 13. CREATE OWUSU'S KUBECONFIG

# Configure the Kubernetes cluster endpoint and embed the CA certificate.
kubectl config set-cluster rag-cluster \
  --server="https://${CONTROL_PLANE_EIP}:6443" \
  --certificate-authority=/etc/kubernetes/pki/ca.crt \
  --embed-certs=true \
  --kubeconfig=owusu.kubeconfig


# Add Owusu's certificate and private key to the kubeconfig.
kubectl config set-credentials owusu \
  --client-certificate=owusu.crt \
  --client-key=owusu.key \
  --embed-certs=true \
  --kubeconfig=owusu.kubeconfig


# Create a context that automatically uses the rag namespace.
kubectl config set-context owusu@rag-cluster \
  --cluster=rag-cluster \
  --user=owusu \
  --namespace=rag \
  --kubeconfig=owusu.kubeconfig


# Set Owusu's context as the default context in this kubeconfig.
kubectl config use-context owusu@rag-cluster \
  --kubeconfig=owusu.kubeconfig


# 14. VERIFY THE GENERATED KUBECONFIG

# Show the context stored in Owusu's kubeconfig.
kubectl config current-context \
  --kubeconfig=owusu.kubeconfig

# Test that Owusu can communicate with the cluster.
kubectl --kubeconfig=owusu.kubeconfig get pods

# Test access to Deployments.
kubectl --kubeconfig=owusu.kubeconfig get deployments

# Test that Secret access is denied.
kubectl --kubeconfig=owusu.kubeconfig get secrets


# 15. VERIFY THE USER IDENTITY

# Show which Kubernetes identity the certificate authenticates as.
kubectl auth whoami \
  --kubeconfig=owusu.kubeconfig




# This is the only file Owusu needs.
ls -lh owusu.kubeconfig
